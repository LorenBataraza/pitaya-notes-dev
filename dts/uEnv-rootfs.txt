# uEnv-rootfs.txt -- U-Boot environment for ext4 root filesystem boot
#
# Memory layout:
#
#   0x2000000  +------------+
#              | rootfs.dtb |  <- device tree blob
#   0x2008000  +------------+
#              | zImage.bin |  <- ARM Linux kernel (zImage format)
#              |            |
#              +------------+
#
# U-Boot loads this file from the FAT partition at startup and evaluates
# bootcmd. To change boot arguments or load addresses, edit this file and
# copy it to the SD card FAT partition as uEnv.txt. No recompilation needed.

dtbaddr=0x2000000
loadaddr=0x2008000
bootargs=console=ttyPS0,115200 earlycon earlyprintk root=/dev/mmcblk0p2 ro rootfstype=ext4 rootwait cma=36M
bootcmd=fatload mmc 0 ${dtbaddr} rootfs.dtb; fatload mmc 0 ${loadaddr} zImage.bin; bootz ${loadaddr} - ${dtbaddr}
